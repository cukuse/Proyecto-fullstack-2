import { Routes, Route } from 'react-router-dom';

import Perfil from './pages/Perfil';

const PRODUCTOS_INICIALES = [
  { id: 1, titulo: "Boda al aire libre", categoria: "matrimonios", aforo: "Aforo: 50 personas", precio: 150000, imagen: "/img/boda-catalgo.jpg", oferta: false },

const VENTAS_INICIALES = [

const USUARIOS_INICIALES = [
  { id: 1, nombre: 'Administrador', email: 'admin@tudia.cl', rol: 'admin', telefono: '+56 9 9999 8888', direccion: 'Oficina Central #100', foto: 'https://images.unsplash.com/photo-1534528741775-53994a69daeb?w=150' },

];

const HISTORIAL_INICIAL = [
  { id: 1, usuario: 'Sistema', accion: 'Inicialización de la plataforma', fecha: '2026-03-01 10:00:00', rol: 'admin' }
];

function App() {
  const [carrito, setCarrito] = useState(() => {
    const guardado = localStorage.getItem('carrito_tu_dia');
    return guardado ? JSON.parse(guardado) : [];
  });

  const [usuario, setUsuario] = useState(() => {
    const usuarioGuardado = localStorage.getItem('usuario_tu_dia');
    return usuarioGuardado ? JSON.parse(usuarioGuardado) : null;
  });

  const [usuarios, setUsuarios] = useState(() => {
    const guardados = localStorage.getItem('usuarios_tu_dia');
    return guardados ? JSON.parse(guardados) : USUARIOS_INICIALES;
  });

  const [productos, setProductos] = useState(() => {
    const guardados = localStorage.getItem('productos_tu_dia');
    return guardados ? JSON.parse(guardados) : PRODUCTOS_INICIALES;
  });

  const [ventas, setVentas] = useState(() => {
    const guardadas = localStorage.getItem('ventas_tu_dia');
    return guardadas ? JSON.parse(guardadas) : VENTAS_INICIALES;
  });

  const [historial, setHistorial] = useState(() => {
    const guardado = localStorage.getItem('historial_tu_dia');
    return guardado ? JSON.parse(guardado) : HISTORIAL_INICIAL;
  });

  useEffect(() => {
    localStorage.setItem('carrito_tu_dia', JSON.stringify(carrito));
  }, [carrito]);

  useEffect(() => {
    if (usuario) {
      localStorage.setItem('usuario_tu_dia', JSON.stringify(usuario));
    } else {
      localStorage.removeItem('usuario_tu_dia');
    }
  }, [usuario]);

  useEffect(() => {
    localStorage.setItem('usuarios_tu_dia', JSON.stringify(usuarios));
  }, [usuarios]);

  useEffect(() => {
    localStorage.setItem('productos_tu_dia', JSON.stringify(productos));
  }, [productos]);

  useEffect(() => {
    localStorage.setItem('ventas_tu_dia', JSON.stringify(ventas));
  }, [ventas]);

  useEffect(() => {
    localStorage.setItem('historial_tu_dia', JSON.stringify(historial));
  }, [historial]);

  const registrarActividad = (accion, rolEjecutor = 'cliente') => {
    const nuevoRegistro = {
      id: Date.now(),
      usuario: usuario ? usuario.email : 'Anónimo',
      accion,
      fecha: new Date().toLocaleString('es-CL'),
      rol: usuario ? usuario.rol : rolEjecutor
    };
    setHistorial(prev => [nuevoRegistro, ...prev]);
  };

  const agregarAlCarrito = (producto) => {
    setCarrito(prev => [...prev, producto]);
    registrarActividad(`Producto "${producto.titulo}" agregado al carrito`);
  };

  const loginUsuario = (datosUsuario) => {
    setUsuario(datosUsuario);
    registrarActividad(`Inicio de sesión de ${datosUsuario.nombre}`, datosUsuario.rol);
  };

  const logoutUsuario = () => {
    if (usuario) {
      registrarActividad(`Cierre de sesión de ${usuario.nombre}`, usuario.rol);
    }
    setUsuario(null);
  };

  return (
    <>
      <Cabecera usuario={usuario} logout={logoutUsuario} />
      <Navegacion totalItems={carrito.length} usuario={usuario} />

      <Routes>
        <Route path="/" element={<Home />} />
        <Route path="/contacto" element={<Contacto />} />
        <Route path="/Nosotros" element={<Nosotros />} />
        <Route path="/Blog" element={<Blog />} />

        <Route 
          path="/catalogo" 
          element={<Catalogo productos={productos} agregarAlCarrito={agregarAlCarrito} />} 
        />
        <Route 
          path="/carrito" 
          element={<Carrito carrito={carrito} setCarrito={setCarrito} />} 
        />
        <Route 
          path="/login" 
          element={<Login usuario={usuario} login={loginUsuario} logout={logoutUsuario} />} 
        />
        <Route 
          path="/perfil" 
          element={
          <Perfil 
        usuario={usuario} 
        setUsuario={setUsuario}
        setUsuarios={setUsuarios}
        registrarActividad={registrarActividad} 
          />}
        />
        <Route 
          path="/admin" 
          element={
            <Admin 
              usuario={usuario}
              productos={productos}
              setProductos={setProductos}
              ventas={ventas}
              setVentas={setVentas}
              usuarios={usuarios}
              setUsuarios={setUsuarios}
              historial={historial}
              registrarActividad={registrarActividad}
            />
          } 
        />
        <Route 
          path="/checkout" 
          element={
            <Checkout 
              carrito={carrito}
              setCarrito={setCarrito}
              setVentas={setVentas}
              registrarActividad={registrarActividad}
              usuario={usuario}
            />
          } 
        />
      </Routes>

      <Pie />
    </>
  );
}

export default App;

import React, { useState } from 'react';

function AdminFormProducto({ onAgregarProducto }) {
  const [titulo, setTitulo] = useState('');
  const [categoria, setCategoria] = useState('matrimonios');
  const [aforo, setAforo] = useState('Aforo: 30 personas');
  const [precio, setPrecio] = useState('');
  const [imagen, setImagen] = useState('/img/boda-catalgo.jpg');
  const [oferta, setOferta] = useState(false);

  const handleSubmit = (e) => {
    e.preventDefault();

    if (!titulo || !precio) {
      alert('Por favor completa el nombre y el precio.');
      return;
    }

    const nuevoProducto = {
      id: Date.now(),
      titulo,
      categoria,
      aforo,
      precio: Number(precio),
      imagen,
      oferta
    };

    onAgregarProducto(nuevoProducto);

    // Limpiar formulario
    setTitulo('');
    setPrecio('');
    setOferta(false);
  };

  return (
    <section className="card shadow-sm mb-4">
      <div className="card-header bg-dark text-white fw-bold">
        ➕ Agregar Nuevo Evento al Catálogo
      </div>
      <div className="card-body">
        <form onSubmit={handleSubmit}>
          {/* row g-3: Fila de Bootstrap con espacio entre campos */}
          <div className="row g-3">
            
            {/* Nombre del Evento */}
            <div className="col-12 col-md-6">
              <label className="form-label fw-bold">Nombre del Evento *</label>
              <input 
                type="text" 
                className="form-control"
                placeholder="Ej. Boda en la Playa"
                value={titulo}
                onChange={(e) => setTitulo(e.target.value)}
                required
              />
            </div>

            {/* Categoría */}
            <div className="col-12 col-md-6">
              <label className="form-label fw-bold">Categoría *</label>
              <select 
                className="form-select"
                value={categoria}
                onChange={(e) => setCategoria(e.target.value)}

                <option value="cumpleanos">🎂 Cumpleaños</option>
                <option value="corporativos">💼 Corporativos</option>
              </select>
            </div>

            {/* Precio */}
            <div className="col-12 col-md-6">
              <label className="form-label fw-bold">Precio ($ CLP) *</label>
              <input

                className="form-control"

import React, { useState } from 'react';

function AdminTablaUsuarios({ usuarios, setUsuarios, registrarActividad }) {
  const [usuarioEditar, setUsuarioEditar] = useState(null);

  const handleGuardarUsuario = (e) => {
    e.preventDefault();
    setUsuarios(prev => prev.map(u => u.id === usuarioEditar.id ? usuarioEditar : u));
    registrarActividad(`Admin modificó la cuenta de: ${usuarioEditar.nombre}`, 'admin');
    setUsuarioEditar(null);
  };

  const handleEliminarUsuario = (id, nombre) => {
    if (window.confirm(`¿Seguro que deseas eliminar al usuario "${nombre}"?`)) {
      setUsuarios(prev => prev.filter(u => u.id !== id));
      registrarActividad(`Admin eliminó al usuario: ${nombre}`, 'admin');
    }
  };

  return (
    <section className="card shadow-sm mb-4 border-0">
      <div className="card-header bg-dark text-white fw-bold d-flex justify-content-between align-items-center">
        <span>👥 Usuarios Registrados ({usuarios.length})</span>
      </div>

      <div className="table-responsive">
        <table className="table table-hover align-middle mb-0">
          <thead className="table-secondary">
            <tr>
              <th>Foto</th>
              <th>Nombre</th>
              <th>Correo</th>
              <th>Rol</th>
              <th>Teléfono</th>
              <th className="text-center">Acciones</th>
            </tr>
          </thead>
          <tbody>
            {usuarios.map((u) => (
              <tr key={u.id}>
                <td>
                  {u.foto ? (
                    <img 
                      src={u.foto} 
                      alt={u.nombre} 
                      className="rounded-circle border"
                      style={{ width: '40px', height: '40px', objectFit: 'cover' }}
                    />
                  ) : (
                    <div className="rounded-circle bg-secondary text-white d-flex align-items-center justify-content-center" style={{ width: '40px', height: '40px' }}>
                      👤
                    </div>
                  )}
                </td>
                <td className="fw-bold">{u.nombre}</td>
                <td>{u.email}</td>
                <td>
                  <span className={`badge ${u.rol === 'admin' ? 'bg-primary' : 'bg-info text-dark'}`}>
                    {u.rol.toUpperCase()}
                  </span>
                </td>
                <td>{u.telefono || 'Sin registro'}</td>
                <td className="text-center">
                  <button 
                    className="btn btn-sm btn-outline-primary me-2"
                    onClick={() => setUsuarioEditar(u)}
                  >
                    ✏️ Editar
                  </button>
                  <button 
                    className="btn btn-sm btn-outline-danger"
                    onClick={() => handleEliminarUsuario(u.id, u.nombre)}
                  >
                    🗑️ Eliminar
                  </button>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>

      {/* MODAL DE EDICIÓN DE USUARIO POR ADMIN */}
      {usuarioEditar && (
        <div className="modal show d-block" style={{ backgroundColor: 'rgba(0,0,0,0.5)' }}>
          <div className="modal-dialog modal-dialog-centered">
            <div className="modal-content">
              <div className="modal-header bg-dark text-white">
                <h5 className="modal-title">✏️ Editar Usuario (Admin)</h5>
                <button type="button" className="btn-close btn-close-white" onClick={() => setUsuarioEditar(null)}></button>
              </div>
              <form onSubmit={handleGuardarUsuario}>
                <div className="modal-body">
                  <div className="mb-3">
                    <label className="form-label fw-bold">Nombre</label>
                    <input 
                      type="text" 
                      className="form-control" 
                      value={usuarioEditar.nombre} 
                      onChange={(e) => setUsuarioEditar({ ...usuarioEditar, nombre: e.target.value })}
                      required 
                    />
                  </div>
                  <div className="mb-3">
                    <label className="form-label fw-bold">Correo Electrónico</label>
                    <input 
                      type="email" 
                      className="form-control" 
                      value={usuarioEditar.email} 
                      onChange={(e) => setUsuarioEditar({ ...usuarioEditar, email: e.target.value })}
                      required 
                    />
                  </div>
                  <div className="mb-3">
                    <label className="form-label fw-bold">URL de Foto de Perfil</label>
                    <input 
                      type="url" 
                      className="form-control" 
                      value={usuarioEditar.foto || ''} 
                      onChange={(e) => setUsuarioEditar({ ...usuarioEditar, foto: e.target.value })}
                    />
                  </div>
                  <div className="mb-3">
                    <label className="form-label fw-bold">Rol</label>
                    <select 
                      className="form-select" 
                      value={usuarioEditar.rol}
                      onChange={(e) => setUsuarioEditar({ ...usuarioEditar, rol: e.target.value })}
                    >
                      <option value="cliente">Cliente</option>
                      <option value="admin">Administrador</option>
                    </select>
                  </div>
                  <div className="mb-3">
                    <label className="form-label fw-bold">Teléfono</label>
                    <input 
                      type="text"

document.addEventListener('DOMContentLoaded', function() {
    const formEvento = document.getElementById('form-evento');
    
    if (formEvento) { // Solo se ejecuta si estamos en evento.html
        formEvento.addEventListener('submit', function(evento) {
            evento.preventDefault();
            let valido = true;

            // 1. Capturar todos los elementos (Incluyendo la nueva fecha)
            const invitados = document.getElementById('evento-invitados');
            const telefono = document.getElementById('evento-telefono');
            const direccion = document.getElementById('evento-direccion');
            const fecha = document.getElementById('evento-fecha'); // Nuevo
            
            const errInvitados = document.getElementById('error-invitados');
            const errTelefono = document.getElementById('error-telefono');
            const errDireccion = document.getElementById('error-direccion');
            const errFecha = document.getElementById('error-fecha'); // Nuevo

            // 2. Limpiar errores anteriores
            invitados.classList.remove('input-error'); 
            errInvitados.style.display = 'none';
            
            telefono.classList.remove('input-error'); 
            errTelefono.style.display = 'none';
            
            direccion.classList.remove('input-error'); 
            errDireccion.style.display = 'none';       
            
            fecha.classList.remove('input-error'); // Nuevo
            errFecha.style.display = 'none';       // Nuevo

            // 3. Validar que invitados sea un número mayor a 0
            if (isNaN(invitados.value) || invitados.value <= 0) {
                errInvitados.textContent = 'Error: Ingresa una cantidad válida de invitados.';
                errInvitados.style.display = 'block';
                invitados.classList.add('input-error');
                valido = false;
            }

            // 4. Validar teléfono (+56 9 seguido de 8 números)
            const regexTelefono = /^\+56\s9\s\d{4}\s\d{4}$/;
            if (!regexTelefono.test(telefono.value.trim())) {
                errTelefono.textContent = 'Sugerencia: Usa el formato +56 9 XXXX XXXX';
                errTelefono.style.display = 'block';
                telefono.classList.add('input-error');
                valido = false;
            }

            // 5. Validar dirección 
            if (direccion.value.trim() === '') {
                errDireccion.textContent = 'Error: La dirección del evento es obligatoria.';
                errDireccion.style.display = 'block';
                direccion.classList.add('input-error');
                valido = false;
            } else if (direccion.value.trim().length < 10) {
                errDireccion.textContent = 'Sugerencia: Ingresa una dirección más detallada (calle y número).';
                errDireccion.style.display = 'block';
                direccion.classList.add('input-error');
                valido = false;
            }

            // 6. Validar fecha (El bloque nuevo)
            if (fecha.value === '') {
                errFecha.textContent = 'Error: Debes seleccionar una fecha en el calendario.';
                errFecha.style.display = 'block';
                fecha.classList.add('input-error');
                valido = false;
            }

import React, { useState } from 'react';

import AdminStats from '../components/AdminStats';
import AdminFormProducto from '../components/AdminFormProducto';
import AdminTablaProductos from '../components/AdminTablaProductos';
import AdminTablaUsuarios from '../components/AdminTablaUsuarios';
import AdminTablaVentas from '../components/AdminTablaVentas';
import AdminBitacora from '../components/AdminBitacora';

function Admin({ usuario, productos, setProductos, ventas, setVentas, usuarios, setUsuarios, historial, registrarActividad }) {
  const navigate = useNavigate();
  const [pestana, setPestana] = useState('kpis'); // 'kpis' | 'productos' | 'usuarios' | 'ventas' | 'bitacora'

  if (!usuario || usuario.rol !== 'admin') {
    return (
      <div className="container text-center py-5">
        <div className="alert alert-danger shadow-sm d-inline-block p-4">
          <h4>🚫 Acceso Restringido</h4>
          <p className="mb-3">Necesitas iniciar sesión con cuenta de Administrador.</p>
          <button className="btn btn-dark" onClick={() => navigate('/login')}>
            Ir a Iniciar Sesión
          </button>
        </div>
      </div>
    );
  }

  const totalRecaudado = ventas.reduce((acc, v) => acc + Number(v.monto || 0), 0);

  const handleAgregarProducto = (nuevoProd) => {
    setProductos([nuevoProd, ...productos]);
    registrarActividad(`Admin agregó el producto: "${nuevoProd.titulo}" ($${nuevoProd.precio})`, 'admin');
    alert('¡Producto agregado exitosamente!');
  };

  const handleEliminarProducto = (id, titulo) => {
    if (window.confirm(`¿Seguro que deseas eliminar "${titulo}"?`)) {
      setProductos(productos.filter(p => p.id !== id));
      registrarActividad(`Admin eliminó el producto: "${titulo}"`, 'admin');
    }
  };

  return (
    <main className="container py-4">
      <div className="d-flex justify-content-between align-items-center mb-4">
        <div>
          <h2 className="fw-bold mb-0">⚙️ Panel Multi-Tareas Admin</h2>
          <small className="text-muted">Gestión de usuarios, órdenes, boletas e inventario</small>
        </div>
        <span className="badge bg-dark fs-6 px-3 py-2">
          👤 {usuario.nombre} ({usuario.rol})
        </span>
      </div>

      {/* MENÚ DE SECCIONES (TABS) */}
      <div className="btn-group w-100 mb-4" role="group">
        <button 
          className={`btn ${pestana === 'kpis' ? 'btn-dark' : 'btn-outline-dark'}`}
          onClick={() => setPestana('kpis')}
        >
          📊 Métricas
        </button>
        <button 
          className={`btn ${pestana === 'usuarios' ? 'btn-dark' : 'btn-outline-dark'}`}
          onClick={() => setPestana('usuarios')}
        >
          👥 Usuarios
        </button>
        <button 
          className={`btn ${pestana === 'ventas' ? 'btn-dark' : 'btn-outline-dark'}`}
          onClick={() => setPestana('ventas')}
        >
          🧾 Órdenes y Boletas
        </button>
        <button 
          className={`btn ${pestana === 'productos' ? 'btn-dark' : 'btn-outline-dark'}`}
          onClick={() => setPestana('productos')}
        >
          📦 Catálogo
        </button>
        <button 
          className={`btn ${pestana === 'bitacora' ? 'btn-dark' : 'btn-outline-dark'}`}
          onClick={() => setPestana('bitacora')}
        >
          📜 Bitácora
        </button>
      </div>

      {/* CONTENIDO SEGÚN LA PESTAÑA SELECCIONADA */}
      {pestana === 'kpis' && (
        <AdminStats 
          totalRecaudado={totalRecaudado} 
          totalProductos={productos.length} 
          totalVentas={ventas.length} 
        />
      )}

      {pestana === 'usuarios' && (
        <AdminTablaUsuarios 
          usuarios={usuarios} 
          setUsuarios={setUsuarios} 
          registrarActividad={registrarActividad} 
        />
      )}

      {pestana === 'ventas' && (

          <AdminTablaProductos 
            productos={productos} 
            onEliminarProducto={handleEliminarProducto} 
          />
        </>
      )}

      {pestana === 'bitacora' && (
        <AdminBitacora historial={historial} />
      )}
    </main>
  );
}

import React, { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import '../css/estilo-login.css';

function Login({ usuario, login, logout }) {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');
  const navigate = useNavigate();

  if (usuario) {
    return (
      <div className="login-page-container">
        <main className="main-centrado login-card" style={{ padding: '50px 20px', textAlign: 'center' }}>
          <h2>¡Hola, {usuario.nombre}!</h2>
          <p style={{ margin: '15px 0' }}>Has iniciado sesión como: <strong>{usuario.rol.toUpperCase()}</strong></p>
          
          {usuario.rol === 'admin' && (
            <button 
              onClick={() => navigate('/admin')} 
              className="btn-accion" 
              style={{ marginBottom: '15px', backgroundColor: '#2a434d' }}
            >
              ⚙️ Ir al Dashboard Admin
            </button>
          )}
          
          <br />
          <button onClick={logout} className="btn-accion" style={{ backgroundColor: '#d9745f' }}>
            Cerrar Sesión
          </button>
        </main>
      </div>
    );
  }

  const handleSubmit = (e) => {
    e.preventDefault();
    setError('');

    const emailL = email.trim().toLowerCase();
    const passL = password.trim();

    if (!emailL || !passL) {
      setError('Por favor ingresa correo y contraseña.');
      return;
    }

    if (emailL === 'admin@tudia.cl' && passL === 'admin123') {
      login({ email: emailL, nombre: 'Administrador', rol: 'admin' });
      alert('¡Bienvenido!');
      navigate('/admin');
    } else {
      login({ email: emailL, nombre: emailL.split('@')[0], rol: 'cliente' });
      alert('¡Bienvenido!');
      navigate('/');
    }
  };

  return (
    <div className="login-page-container">
      <main className="main-centrado login-card">
        <div className="cabecilla-login">
          <div className="icono-login">
            <img src="/img/logo2.png" alt="Logo Tu Día Especial" style={{ width: '120px' }} />
          </div>
        </div>

        <form className="caja-formulario" onSubmit={handleSubmit}>
          <h2 style={{ color: 'white', textAlign: 'center', marginBottom: '15px' }}>Iniciar Sesión</h2>

          {error && <p style={{ color: '#ff6b6b', fontWeight: 'bold', textAlign: 'center' }}>{error}</p>}

          <div className="grupo-input">
            <label>Correo Electrónico</label>
            <input
              type="email"
              placeholder="admin@tudia.cl o cliente@tudia.cl"
              value={email}
              onChange={(e) => setEmail(e.target.value)}
              required
            />
          </div>

          <div className="grupo-input">
            <label>Contraseña</label>
            <input
              type="password"
              placeholder="admin123 o cliente123"
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              required
            />
          </div>

          <button type="submit" className="btn-accion">
            Ingresar
          </button>

          <a 
            href="#olvide" 
            onClick={(e) => { e.preventDefault(); alert('Instrucciones enviadas a tu correo.'); }}

import React, { useState } from 'react';
import { useNavigate } from 'react-router-dom';

function Perfil({ usuario, setUsuario, setUsuarios, registrarActividad }) {
  const navigate = useNavigate();

  if (!usuario) {
    return (
      <div className="container text-center py-5">
        <div className="alert alert-warning shadow-sm d-inline-block p-4">
          <h4>🔒 Sesión no iniciada</h4>
          <p>Debes iniciar sesión para ver y editar tu perfil.</p>
          <button className="btn btn-dark" onClick={() => navigate('/login')}>
            Ir a Iniciar Sesión
          </button>
        </div>
      </div>
    );
  }

  const [nombre, setNombre] = useState(usuario.nombre || '');
  const [email, setEmail] = useState(usuario.email || '');
  const [telefono, setTelefono] = useState(usuario.telefono || '+56 9 1234 5678');
  const [direccion, setDireccion] = useState(usuario.direccion || 'Av. Providencia 1234, Santiago');
  const [foto, setFoto] = useState(usuario.foto || '');
  const [mensajeGuardado, setMensajeGuardado] = useState(false);

  // Convertir imagen seleccionada a Base64
  const handleSubirFotoLocal = (e) => {
    const file = e.target.files[0];
    if (file) {
      if (file.size > 2 * 1024 * 1024) {
        alert("La imagen debe ser menor a 2MB.");
        return;
      }
      const reader = new FileReader();
      reader.onloadend = () => {
        setFoto(reader.result);
      };
      reader.readAsDataURL(file);
    }
  };

  const handleSubmit = (e) => {
    e.preventDefault();

    const usuarioActualizado = {
      ...usuario,
      nombre,
      email,
      telefono,
      direccion,
      foto
    };

    // Actualiza sesión actual y listado general
    setUsuario(usuarioActualizado);
    if (setUsuarios) {
      setUsuarios(prev => prev.map(u => u.id === usuario.id ? usuarioActualizado : u));
    }

    registrarActividad(`Usuario ${usuario.nombre} actualizó su perfil y foto`, usuario.rol);

    setMensajeGuardado(true);
    setTimeout(() => setMensajeGuardado(false), 3000);
  };

  return (
    <main className="container py-4" style={{ maxWidth: '700px' }}>
      <div className="card shadow border-0">
        <div className="card-header bg-dark text-white d-flex justify-content-between align-items-center py-3">
          <h4 className="mb-0 fw-bold">👤 Mi Perfil de Usuario</h4>
          <span className={`badge ${usuario.rol === 'admin' ? 'bg-primary' : 'bg-info text-dark'}`}>
            {usuario.rol.toUpperCase()}
          </span>
        </div>

        <div className="card-body p-4">
          {mensajeGuardado && (
            <div className="alert alert-success alert-dismissible fade show mb-4" role="alert">
              ✅ <strong>¡Perfil actualizado con éxito!</strong> Los cambios han sido guardados.
            </div>
          )}

          {/* ÁREA DE FOTO DE PERFIL (SOLO SUBIR ARCHIVO) */}
          <div className="text-center mb-4">
            <div className="position-relative d-inline-block">
              {foto ? (
                <img 
                  src={foto} 
                  alt="Foto de Perfil" 
                  className="rounded-circle shadow border border-3 border-danger"
                  style={{ width: '130px', height: '130px', objectFit: 'cover' }}
                />
              ) : (
                <div 
                  className="rounded-circle bg-secondary text-white d-flex align-items-center justify-content-center shadow border border-3 border-danger mx-auto"
                  style={{ width: '130px', height: '130px', fontSize: '3.5rem' }}
                >
                  👤
                </div>
              )}
            </div>

            <div className="mt-3">
              <div className="d-flex justify-content-center gap-2">
                <label className="btn btn-dark fw-bold px-3">
                  📁 Subir Imagen
                  <input type="file" accept="image/*" hidden onChange={handleSubirFotoLocal} />
                </label>
                {foto && (
                  <button type="button" className="btn btn-outline-danger" onClick={() => setFoto('')}>
                    🗑️ Eliminar Foto
                  </button>
                )}
              </div>
              <small className="text-muted d-block mt-2">Archivos admitidos: JPG, PNG (Máx. 2MB)</small>
            </div>
          </div>

          <form onSubmit={handleSubmit}>
            <div className="mb-3">
              <label className="form-label fw-bold">Nombre Completo</label>
              <input 
                type="text" 
                className="form-control" 
                value={nombre} 
                onChange={(e) => setNombre(e.target.value)}
                required 
              />
            </div>

            <div className="mb-3">
              <label className="form-label fw-bold">Correo Electrónico</label>
              <input 
                type="email" 
                className="form-control" 
                value={email} 
                onChange={(e) => setEmail(e.target.value)}
                required 
              />
            </div>

            <div className="mb-3">
              <label className="form-label fw-bold">Teléfono de Contacto</label>
              <input 
                type="text" 
                className="form-control" 
                value={telefono} 
                onChange={(e) => setTelefono(e.target.value)}
                required 
              />
            </div>

            <div className="mb-3">
              <label className="form-label fw-bold">Dirección Principal</label>
              <input 
                type="text" 
                className="form-control" 
                value={direccion} 
                onChange={(e) => setDireccion(e.target.value)}
                required 
              />
            </div>

            <div className="d-grid gap-2 mt-4">
              <button type="submit" className="btn btn-danger fw-bold py-2 fs-5">
                💾 Guardar Cambios
              </button>
              {usuario.rol === 'admin' && (
                <button type="button" className="btn btn-outline-dark" onClick={() => navigate('/admin')}>
                  ⚙️ Ir al Panel de Administración
                </button>
              )}
            </div>
          </form>
        </div>
      </div>
    </main>
  );
}

export default Perfil;

import React from 'react';
import { Link, useNavigate } from 'react-router-dom';
import Boton from '../components/Boton';
import '../css/catalogo.css';

function Carrito({ carrito = [], setCarrito }) {
    const navigate = useNavigate();
    const cantidadEventos = carrito.length;
    const totalPagar = carrito.reduce((suma, evento) => suma + Number(evento.precio || 0), 0);

    const handleVaciar = () => {
        if (setCarrito) setCarrito([]);
    };

    const handleEliminarItem = (indexAEliminar) => {
        if (setCarrito) {
            setCarrito(carrito.filter((_, index) => index !== indexAEliminar));
        }
    };

    return (
        <main className="main-carrito">
            <h2 className="titulo-seccion">Detalle de tu Carrito</h2>

            <div className="contenedor-carrito">
                <div className="lista-carrito">
                    {carrito.length === 0 ? (
                        <p className="carrito-vacio-msg">No tienes eventos en tu carrito aún.</p>
                    ) : (
                        carrito.map((item, index) => (
                            <div key={index} className="item-carrito" style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: '10px' }}>
                                <div>
                                    <h4>{item.titulo}</h4>
                                    <p>${Number(item.precio).toLocaleString('es-CL')}</p>
                                </div>
                                <Boton onClick={() => handleEliminarItem(index)}>
                                    ❌
                                </Boton>
                            </div>
                        ))
                    )}
                </div>

                <div className="resumen-compra">
                    <h3>Resumen de Reserva</h3>
                    
                    <div className="linea-resumen">
                        <span>Cantidad de eventos:</span>
                        <span>{cantidadEventos}</span>
                    </div>

                    <div className="linea-resumen total">
                        <span>Total a Pagar:</span>
                        <strong>${totalPagar.toLocaleString('es-CL')}</strong>
                    </div>
                    
                    {/* Redirección al checkout */}
                    <button 
                        className="btn-pagar" 
                        disabled={carrito.length === 0}
                        onClick={() => navigate('/checkout')}
                    >
                        Proceder al Pago
                    </button>
                    
                    <Boton className="btn-vaciar" onClick={handleVaciar}>
                        Vaciar Carrito
                    </Boton>

                    <Link to="/catalogo" className="link-volver">
                        ← Volver al Catálogo
                    </Link>
                </div>
            </div>
        </main>
    );
}

export default Carrito;

import React, { useState } from 'react';
import TarjetaEvento from '../components/TarjetaEvento';
import FiltroCatalogo from '../components/FiltroCatalogo';
import '../css/estilo.css';
import '../css/catalogo.css';

function Catalogo({ productos = [], agregarAlCarrito }) {
    const [busqueda, setBusqueda] = useState('');
    const [categoriaSeleccionada, setCategoriaSeleccionada] = useState('todas');

    const eventosFiltrados = productos.filter((evento) => {
        const coincideCategoria = 
            categoriaSeleccionada === 'todas' ||
            (categoriaSeleccionada === 'ofertas' && evento.oferta) ||
            evento.categoria === categoriaSeleccionada;

        const coincideBusqueda = evento.titulo
            .toLowerCase()
            .includes(busqueda.toLowerCase());

        return coincideCategoria && coincideBusqueda;
    });

    return (
        <main className="main-catalogo">
            <h2 className="titulo-seccion">Nuestros Eventos</h2>

            <FiltroCatalogo 
                busqueda={busqueda} 
                setBusqueda={setBusqueda} 
                categoriaSeleccionada={categoriaSeleccionada} 
                setCategoriaSeleccionada={setCategoriaSeleccionada} 
            />

            <div className="grilla-catalogo">
                {eventosFiltrados.length === 0 ? (
                    <p style={{ textAlign: 'center', width: '100%', padding: '30px' }}>
                        🔍 No se encontraron eventos que coincidan con tu búsqueda.
                    </p>
                ) : (
                    eventosFiltrados.map((evento) => (
                        <TarjetaEvento 
                            key={evento.id} 
                            evento={evento} 
                            agregarAlCarrito={agregarAlCarrito} 
                        />
                    ))
                )}
            </div>
        </main>
    );
}

export default Catalogo;

    // 2. Modificar cantidades (+ / -)
    window.cambiarCantidad = function(id, cambio) {
        const item = carrito.find(p => p.id === id);
        if (item) {
            item.cantidad += cambio;
            if (item.cantidad <= 0) {
                carrito = carrito.filter(p => p.id !== id);
            }
            guardarYActualizar();
        }
    };

    // 3. Eliminar evento individual
    window.eliminarItem = function(id) {
        carrito = carrito.filter(p => p.id !== id);
        guardarYActualizar();
    };

    // 4. Vaciar carrito completo
    if (btnVaciar) {
        btnVaciar.addEventListener('click', () => {
            if (confirm('¿Seguro que deseas vaciar el carrito?')) {
                carrito = [];
                guardarYActualizar();
            }
        });
    }

    // 5. Guardar en localStorage y refrescar la vista
    function guardarYActualizar() {
        localStorage.setItem('carrito_tu_dia', JSON.stringify(carrito));
        actualizarContadorHeader();
        if (listaCarritoEl) {
            renderizarPaginaCarrito();
        }
    }

    function actualizarContadorHeader() {
        if (contadorBadge) {
            const totalItems = carrito.reduce((acc, item) => acc + item.cantidad, 0);
            contadorBadge.textContent = totalItems;
        }
    }

    // 6. Renderizar contenido de carrito.html
    function renderizarPaginaCarrito() {
        listaCarritoEl.innerHTML = '';

        if (carrito.length === 0) {
            listaCarritoEl.innerHTML = `
                <div class="carrito-vacio-msg">
                    <h3>Tu carrito está vacío 🛒</h3>
                    <p>Explora nuestro catálogo para elegir tus eventos preferidos.</p>
                </div>
            `;
            if (resumenCantidadEl) resumenCantidadEl.textContent = '0';
            if (resumenTotalEl) resumenTotalEl.textContent = '$0';
            return;
        }

        carrito.forEach(item => {
            const itemDiv = document.createElement('div');
            itemDiv.classList.add('item-carrito');
            itemDiv.innerHTML = `
                <img src="${item.imagen}" alt="${item.titulo}" class="img-carrito">
                <div class="info-item">
                    <h3>${item.titulo}</h3>
                    <p>${item.aforo}</p>
                    <p><strong>$${item.precio.toLocaleString('es-CL')}</strong> c/u</p>
                    <div class="controles-cantidad">
                        <button class="btn-cant" onclick="cambiarCantidad('${item.id}', -1)">-</button>
                        <span>Cant: <strong>${item.cantidad}</strong></span>
                        <button class="btn-cant" onclick="cambiarCantidad('${item.id}', 1)">+</button>
                    </div>
                </div>
                <div>
                    <p>Subtotal:</p>
                    <strong>$${(item.precio * item.cantidad).toLocaleString('es-CL')}</strong>
                </div>
                <button class="btn-eliminar-item" onclick="eliminarItem('${item.id}')" title="Eliminar">&times;</button>
            `;
            listaCarritoEl.appendChild(itemDiv);
        });

        const totalAcumulado = carrito.reduce((acc, item) => acc + (item.precio * item.cantidad), 0);
        const cantidadTotal = carrito.reduce((acc, item) => acc + item.cantidad, 0);

        if (resumenCantidadEl) resumenCantidadEl.textContent = cantidadTotal;
        if (resumenTotalEl) resumenTotalEl.textContent = `$${totalAcumulado.toLocaleString('es-CL')}`;
    }

    // Inicializar contador al cargar cualquier vista
    actualizarContadorHeader();
    if (listaCarritoEl) {
        renderizarPaginaCarrito();
    }
});

document.addEventListener('DOMContentLoaded', () => {
    const inputBuscar = document.getElementById('input-buscar');
    const btnLimpiar = document.getElementById('btn-limpiar-busqueda');
    const chips = document.querySelectorAll('.btn-chip');
    const tarjetas = document.querySelectorAll('.tarjeta-evento');
    const sinResultados = document.getElementById('sin-resultados');
    const btnRestablecer = document.getElementById('btn-restablecer');

    let categoriaActiva = 'todas';

    // 1. Detectar si viene el parámetro ofertas=true en la URL
    const urlParams = new URLSearchParams(window.location.search);
    if (urlParams.get('ofertas') === 'true') {
        activarChip('ofertas');
    }

    // 2. Eventos en los Chips de Categoría
    chips.forEach(chip => {
        chip.addEventListener('click', () => {
            activarChip(chip.dataset.categoria);
        });
    });

    function activarChip(cat) {
        categoriaActiva = cat;
        chips.forEach(c => {
            if (c.dataset.categoria === cat) {
                c.classList.add('active');
            } else {
                c.classList.remove('active');
            }
        });
        filtrarEventos();
    }

    // 3. Evento de Búsqueda por Texto
    if (inputBuscar) {
        inputBuscar.addEventListener('input', () => {
            if (inputBuscar.value.trim() !== '') {
                btnLimpiar.style.display = 'block';
            } else {
                btnLimpiar.style.display = 'none';
            }
            filtrarEventos();
        });
    }

    if (btnLimpiar) {
        btnLimpiar.addEventListener('click', () => {
            inputBuscar.value = '';
            btnLimpiar.style.display = 'none';
            filtrarEventos();
        });
    }

    if (btnRestablecer) {
        btnRestablecer.addEventListener('click', () => {
            inputBuscar.value = '';
            if (btnLimpiar) btnLimpiar.style.display = 'none';
            activarChip('todas');
        });
    }

    // 4. Función de Filtrado General
    function filtrarEventos() {
        const texto = inputBuscar ? inputBuscar.value.toLowerCase().trim() : '';
        let visibles = 0;

        tarjetas.forEach(tarjeta => {
            const titulo = tarjeta.querySelector('h3').textContent.toLowerCase();
            const categoria = tarjeta.dataset.categoria;
            const esOferta = tarjeta.dataset.oferta === 'true';

            // Comprobación de texto
            const coincideTexto = titulo.includes(texto);

            // Comprobación de categoría
            let coincideCategoria = false;
            if (categoriaActiva === 'todas') {
                coincideCategoria = true;
            } else if (categoriaActiva === 'ofertas') {
                coincideCategoria = esOferta;
            } else {
                coincideCategoria = (categoria === categoriaActiva);
            }

            // Aplicar visibilidad
            if (coincideTexto && coincideCategoria) {
                tarjeta.style.display = 'flex';
                visibles++;
            } else {
                tarjeta.style.display = 'none';
            }
        });

        // Mostrar u ocultar mensaje de "Sin resultados"
        if (sinResultados) {
            sinResultados.style.display = (visibles === 0) ? 'block' : 'none';
        }
    }
});

    const formEvento = document.getElementById('form-evento');

            // 1. Capturar todos los elementos (Incluyendo la nueva fecha)
            const invitados = document.getElementById('evento-invitados');
            const telefono = document.getElementById('evento-telefono');
            const direccion = document.getElementById('evento-direccion');
            const fecha = document.getElementById('evento-fecha'); // Nuevo

            const errTelefono = document.getElementById('error-telefono');
            const errDireccion = document.getElementById('error-direccion');
            const errFecha = document.getElementById('error-fecha'); // Nuevo

            // 2. Limpiar errores anteriores
            invitados.classList.remove('input-error'); 
            errInvitados.style.display = 'none';
            
            telefono.classList.remove('input-error'); 
            errTelefono.style.display = 'none';
            
            direccion.classList.remove('input-error'); 
            errDireccion.style.display = 'none';       
            
            fecha.classList.remove('input-error'); // Nuevo
            errFecha.style.display = 'none';       // Nuevo

            // 3. Validar que invitados sea un número mayor a 0
            if (isNaN(invitados.value) || invitados.value <= 0) {
                errInvitados.textContent = 'Error: Ingresa una cantidad válida de invitados.';
                errInvitados.style.display = 'block';
                invitados.classList.add('input-error');
                valido = false;
            }

            // 4. Validar teléfono (+56 9 seguido de 8 números)
            const regexTelefono = /^\+56\s9\s\d{4}\s\d{4}$/;
            if (!regexTelefono.test(telefono.value.trim())) {
                errTelefono.textContent = 'Sugerencia: Usa el formato +56 9 XXXX XXXX';
                errTelefono.style.display = 'block';
                telefono.classList.add('input-error');
                valido = false;
            }

            // 5. Validar dirección 
            if (direccion.value.trim() === '') {
                errDireccion.textContent = 'Error: La dirección del evento es obligatoria.';
                errDireccion.style.display = 'block';
                direccion.classList.add('input-error');
                valido = false;
            } else if (direccion.value.trim().length < 10) {
                errDireccion.textContent = 'Sugerencia: Ingresa una dirección más detallada (calle y número).';
                errDireccion.style.display = 'block';
                direccion.classList.add('input-error');
                valido = false;
            }

            // 6. Validar fecha (El bloque nuevo)
            if (fecha.value === '') {
                errFecha.textContent = 'Error: Debes seleccionar una fecha en el calendario.';
                errFecha.style.display = 'block';
                fecha.classList.add('input-error');
                valido = false;
            }

            // 7. Ejecutar si todo es válido
            if (valido) {
                alert('¡Evento agendado con éxito!');
                formEvento.reset();
            }
        });
    }
